    // o el centro de la ventana activa. Ese punto se queda clavado en su sitio
    // de la pantalla mientras todo lo demás gira alrededor.
    const cx = (v.ox || 0) + (v.w || w) / 2, cy = (v.oy || 0) + (v.h || h) / 2;
    // Desde la 0.21.4 la Perspectiva orbita alrededor de su **mira** (ver
    // ventanas.js): el punto que fijó el último Extents. Mike (24-sep): «así
    // se puede anclar en un nuevo punto el eje de rotación». Sin mira (una
    // vista sin ventanas), lo de siempre: el pivote dado o el centro.
    const centro = v.mira ? v.mira
      : (pivote ? [pivote[0], pivote[1], 0] : (window.aMM ? [...window.aMM(cx, cy), 0] : null));
    const antes = centro && window.aPX ? window.aPX(centro[0], centro[1], centro[2] || 0) : null;
    v.rx = rx;
    v.rz = rz;
    if (antes) {
      const q = window.aPX(centro[0], centro[1], centro[2] || 0);
      // Ojo con el signo de la Y: en el dibujo crece hacia arriba y en la
      // pantalla hacia abajo, así que la corrección va al revés que la de X.
      v.x += (q[0] - antes[0]) / v.escala;
